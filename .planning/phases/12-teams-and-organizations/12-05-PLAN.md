```typescript
import { api } from "./api";

export type OrgRole = "owner" | "scientist" | "viewer";

export interface OrgResponse {
  id: string;
  name: string;
  role: OrgRole;
  is_personal: boolean;
}

export interface MemberRow {
  user_id: string;
  email: string;
  role: OrgRole;
  created_at: string;
}

export interface InvitationRow {
  id: string;
  email: string;
  role: OrgRole;
  expires_at: string;
  accepted_at: string | null;
  revoked_at: string | null;
}

export interface PreviewResult {
  valid: boolean;
  reason?: "not_found" | "revoked" | "expired" | "already_accepted";
  organization_name?: string;
  role?: OrgRole;
  email?: string;
}

export async function fetchMyOrgs(): Promise<OrgResponse[]> {
  const data = await api<{ orgs: OrgResponse[] }>("/organizations/mine");
  return data.orgs;
}

export async function createOrg(name: string): Promise<OrgResponse> {
  return api<OrgResponse>("/organizations", { method: "POST", body: { name } });
}

export async function fetchMembers(orgId: string): Promise<MemberRow[]> {
  const data = await api<{ members: MemberRow[] }>(`/organizations/${orgId}/members`);
  return data.members;
}

export async function inviteMember(
  orgId: string, email: string, role: OrgRole,
): Promise<{ id: string }> {
  return api(`/organizations/${orgId}/invitations`, {
    method: "POST", body: { email, role },
  });
}

export async function acceptInvitation(
  token: string,
): Promise<{ organization_id: string; role: OrgRole }> {
  return api("/invitations/accept", { method: "POST", body: { token } });
}

export async function previewInvitation(token: string): Promise<PreviewResult> {
  return api(`/invitations/preview?token=${encodeURIComponent(token)}`);
}

export async function transferOwnership(
  orgId: string, targetUserId: string, newSelfRole: "scientist" | "viewer",
): Promise<void> {
  await api(`/organizations/${orgId}/members/transfer`, {
    method: "POST",
    body: { target_user_id: targetUserId, new_self_role: newSelfRole },
  });
}

export async function removeMember(orgId: string, userId: string): Promise<void> {
  await api(`/organizations/${orgId}/members/${userId}`, { method: "DELETE" });
}

export async function updateMemberRole(
  orgId: string, userId: string, role: OrgRole,
): Promise<void> {
  await api(`/organizations/${orgId}/members/${userId}`, {
    method: "PATCH", body: { role },
  });
}

export async function revokeInvitation(orgId: string, inviteId: string): Promise<void> {
  await api(`/organizations/${orgId}/invitations/${inviteId}`, { method: "DELETE" });
}

export async function fetchPendingInvitations(orgId: string): Promise<InvitationRow[]> {
  const data = await api<{ invitations: InvitationRow[] }>(
    `/organizations/${orgId}/invitations?status=pending`,
  );
  return data.invitations;
}

export async function renameOrg(orgId: string, name: string): Promise<OrgResponse> {
  return api(`/organizations/${orgId}`, { method: "PATCH", body: { name } });
}

export async function deleteOrg(orgId: string): Promise<void> {
  await api(`/organizations/${orgId}`, { method: "DELETE" });
}
```

3. Create frontend/src/components/org/OrganizationContext.tsx:

```typescript
import { createContext, useCallback, useContext, useEffect, useState } from "react";
import { fetchMyOrgs, type OrgResponse, type OrgRole } from "@/lib/organizations";

interface OrgContextValue {
  orgs: OrgResponse[];
  activeOrgId: string | null;
  activeOrg: OrgResponse | null;
  role: OrgRole | null;
  loading: boolean;
  refresh: () => Promise<void>;
  setActiveOrg: (orgId: string) => void;
}

const OrgContext = createContext<OrgContextValue | null>(null);

const STORAGE_KEY = "kendrew.activeOrgId";

export function OrgProvider({ children }: { children: React.ReactNode }) {
  const [orgs, setOrgs] = useState<OrgResponse[]>([]);
  const [activeOrgId, setActiveOrgIdState] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);

  const refresh = useCallback(async () => {
    setLoading(true);
    try {
      const list = await fetchMyOrgs();
      setOrgs(list);
      const stored = localStorage.getItem(STORAGE_KEY);
      const validStored = stored && list.some((o) => o.id === stored);
      if (validStored) {
        setActiveOrgIdState(stored);
      } else {
        const personal = list.find((o) => o.is_personal) || list[0];
        if (personal) {
          localStorage.setItem(STORAGE_KEY, personal.id);
          setActiveOrgIdState(personal.id);
        } else {
          localStorage.removeItem(STORAGE_KEY);
          setActiveOrgIdState(null);
        }
      }
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { refresh().catch(console.error); }, [refresh]);

  const setActiveOrg = useCallback((orgId: string) => {
    localStorage.setItem(STORAGE_KEY, orgId);
    window.location.reload();
  }, []);

  const activeOrg = orgs.find((o) => o.id === activeOrgId) ?? null;
  const role = activeOrg?.role ?? null;

  return (
    <OrgContext.Provider value={{ orgs, activeOrgId, activeOrg, role, loading, refresh, setActiveOrg }}>
      {children}
    </OrgContext.Provider>
  );
}

export function useOrgContext(): OrgContextValue {
  const ctx = useContext(OrgContext);
  if (!ctx) throw new Error("useOrgContext must be used inside <OrgProvider>");
  return ctx;
}

export function clearActiveOrgOnLogout() {
  localStorage.removeItem(STORAGE_KEY);
}
```

```tsx
import { useState } from "react";
import { useNavigate } from "react-router-dom";

import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { useOrgContext } from "@/components/org/OrganizationContext";
import { createOrg } from "@/lib/organizations";

export function CreateOrganization() {
  const [name, setName] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const { refresh, setActiveOrg } = useOrgContext();
  const navigate = useNavigate();

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    if (!name.trim()) return;
    setSubmitting(true); setError(null);
    try {
      const org = await createOrg(name.trim());
      await refresh();
      setActiveOrg(org.id);  // triggers reload into the new org
    } catch (err: any) {
      setError(err?.message ?? "Failed to create organization");
      setSubmitting(false);
    }
  }

  return (
    <main className="container max-w-md py-12">
      <h1 className="font-display text-2xl mb-2">Create organization</h1>
      <p className="text-sm text-muted-foreground mb-6">
        Invite teammates and share job history, billing, and results.
      </p>
      <form onSubmit={handleSubmit} className="space-y-4">
        <div>
          <label htmlFor="org-name" className="text-sm font-medium">Organization name</label>
          <Input id="org-name" value={name} onChange={(e) => setName(e.target.value)}
                 maxLength={100} required autoFocus />
        </div>
        {error && <p className="text-sm text-destructive">{error}</p>}
        <div className="flex gap-2">
          <Button type="submit" disabled={submitting || !name.trim()}>
            {submitting ? "Creating..." : "Create organization"}
          </Button>
          <Button type="button" variant="ghost" onClick={() => navigate(-1)}>Cancel</Button>
        </div>
      </form>
    </main>
  );
}

export default CreateOrganization;
```

2. Create frontend/src/pages/AcceptInvitation.tsx covering all four §6.2 branches:

```tsx
import { useEffect, useState } from "react";
import { useNavigate, useSearchParams } from "react-router-dom";

import { Button } from "@/components/ui/button";
import { useOrgContext } from "@/components/org/OrganizationContext";
import { acceptInvitation, previewInvitation, type PreviewResult } from "@/lib/organizations";
import { api } from "@/lib/api";

interface MeResponse { user_id: string; email: string; }

export function AcceptInvitation() {
  const [params] = useSearchParams();
  const token = params.get("token") || "";
  const [preview, setPreview] = useState<PreviewResult | null>(null);
  const [me, setMe] = useState<MeResponse | null | "loading">("loading");
  const [error, setError] = useState<string | null>(null);
  const [accepted, setAccepted] = useState(false);
  const navigate = useNavigate();
  const orgCtx = useOrgContext();

  useEffect(() => {
    if (!token) { setError("Missing invitation token"); return; }
    previewInvitation(token).then(setPreview).catch((e) => setError(e.message));
    api<MeResponse>("/auth/me").then(setMe).catch(() => setMe(null));
  }, [token]);

  if (error) return <Branch>{error}</Branch>;
  if (!preview || me === "loading") return <Branch>Loading invitation...</Branch>;
  if (!preview.valid) return <Branch>{branchInvalidMessage(preview)}</Branch>;

  // Branch: signed out + token valid -> route to login or signup, carrying token.
  if (me === null) {
    return (
      <Branch>
        <p className="text-sm">
          You've been invited to join <strong>{preview.organization_name}</strong> as{" "}
          <strong>{preview.role}</strong>.
        </p>
        <p className="text-sm text-muted-foreground">
          Sign in as <strong>{preview.email}</strong> to accept this invitation.
        </p>
        <div className="flex gap-2 mt-4">
          <Button onClick={() => navigate(`/login?invite_token=${token}&next=/invitations/accept`)}>
            Sign in
          </Button>
          <Button variant="outline" onClick={() => navigate(`/signup?invite_token=${token}`)}>
            Create account
          </Button>
        </div>
      </Branch>
    );
  }

  // Branch: signed in + email mismatch
  if (me.email.toLowerCase() !== (preview.email ?? "").toLowerCase()) {
    return (
      <Branch>
        <p className="text-sm">
          This invitation is for <strong>{preview.email}</strong>.
        </p>
        <p className="text-sm text-muted-foreground">
          You're currently signed in as <strong>{me.email}</strong>.
          Sign out and sign in as the invited email to accept.
        </p>
        <Button onClick={async () => {
          await api("/auth/logout", { method: "POST" });
          navigate(`/login?invite_token=${token}&next=/invitations/accept`);
        }}>Sign out</Button>
      </Branch>
    );
  }

  // Branch: signed in + email matches -> accept button
  if (accepted) return <Branch>Joined! Redirecting...</Branch>;

  return (
    <Branch>
      <p className="text-sm">
        Join <strong>{preview.organization_name}</strong> as <strong>{preview.role}</strong>?
      </p>
      <Button onClick={async () => {
        try {
          const result = await acceptInvitation(token);
          await orgCtx.refresh();
          orgCtx.setActiveOrg(result.organization_id);  // reloads into the new org
          setAccepted(true);
        } catch (e: any) {
          setError(e?.message ?? "Failed to accept invitation");
        }
      }}>Accept invitation</Button>
    </Branch>
  );
}

function Branch({ children }: { children: React.ReactNode }) {
  return <main className="container max-w-md py-12 space-y-3">{children}</main>;
}

function branchInvalidMessage(p: PreviewResult): string {
  switch (p.reason) {
    case "expired": return `This invitation expired. Ask the organization owner to send a new one.`;
    case "revoked": return `This invitation was revoked.`;
    case "already_accepted": return `This invitation has already been accepted.`;
    default: return `Invitation not found. The link may be invalid.`;
  }
}

export default AcceptInvitation;
```
