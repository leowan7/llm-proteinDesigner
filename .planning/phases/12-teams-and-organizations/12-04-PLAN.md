```python
"""Phase 12: Stamp organization metadata onto migrated Stripe customers.

One-shot script. Run AFTER Plan 12-01's migration has moved stripe_customer_id
from public.users to public.organizations, AND AFTER Plan 12-03's backend cutover
deploys. Run BEFORE Plan 12-06's drop of public.users.stripe_customer_id.

Usage:
    # Smoke test in Stripe test mode
    STRIPE_TEST_SECRET_KEY=sk_test_... python backend/scripts/stamp_stripe_org_metadata.py --test-mode --dry-run

    # Real test-mode run
    STRIPE_TEST_SECRET_KEY=sk_test_... python backend/scripts/stamp_stripe_org_metadata.py --test-mode

    # Production dry-run
    python backend/scripts/stamp_stripe_org_metadata.py --dry-run

    # Production live run (set DATABASE_URL + STRIPE_SECRET_KEY)
    python backend/scripts/stamp_stripe_org_metadata.py
"""

from __future__ import annotations

import argparse
import asyncio
import json
import logging
import os
import sys
import time
from datetime import date
from typing import Any

import asyncpg
import stripe

from config import settings


logger = logging.getLogger("stamp_stripe_org_metadata")


MIGRATION_DATE = date.today().isoformat()  # stamped into metadata for traceability
MAX_RETRIES = 3


def _resolve_stripe_key(test_mode: bool) -> str:
    if test_mode:
        key = os.environ.get("STRIPE_TEST_SECRET_KEY", "")
        if not key:
            raise SystemExit("ERROR: --test-mode requires STRIPE_TEST_SECRET_KEY env var")
        return key
    if not settings.stripe_secret_key:
        raise SystemExit("ERROR: STRIPE_SECRET_KEY not configured")
    return settings.stripe_secret_key


def _build_metadata(org_id: str, name: str, is_personal: bool) -> dict[str, str]:
    return {
        "organization_id": str(org_id),
        "kendrew_org_name": name,
        "is_personal": "true" if is_personal else "false",
        "migrated_from_user_v1": MIGRATION_DATE,
    }


def _is_already_tagged(existing_meta: dict, target_meta: dict) -> bool:
    """Returns True if the Stripe customer already carries our org metadata."""
    return existing_meta.get("organization_id") == target_meta["organization_id"]


async def _process_org(row: asyncpg.Record, dry_run: bool) -> dict[str, Any]:
    """Process one org row; returns the outcome dict."""
    target_meta = _build_metadata(row["id"], row["name"], row["is_personal"])
    customer_id = row["stripe_customer_id"]
    out: dict[str, Any] = {
        "org_id": str(row["id"]),
        "name": row["name"],
        "customer_id": customer_id,
    }
    if dry_run:
        out["outcome"] = "would-modify"
        out["target_metadata"] = target_meta
        return out

    for attempt in range(MAX_RETRIES):
        try:
            existing = stripe.Customer.retrieve(customer_id)
            if _is_already_tagged(existing.metadata or {}, target_meta):
                out["outcome"] = "skipped-already-tagged"
                return out
            stripe.Customer.modify(customer_id, metadata=target_meta)
            out["outcome"] = "modified"
            return out
        except stripe.error.RateLimitError:
            wait = 2 ** attempt
            logger.warning("Rate limit on %s; sleeping %ds", customer_id, wait)
            time.sleep(wait)
        except stripe.error.StripeError as exc:
            out["outcome"] = "failed"
            out["error"] = str(exc)
            return out
    out["outcome"] = "failed"
    out["error"] = "rate limit retries exhausted"
    return out


async def main_async(args: argparse.Namespace) -> int:
    stripe.api_key = _resolve_stripe_key(args.test_mode)
    pool = await asyncpg.create_pool(settings.database_url, min_size=1, max_size=2)
    try:
        sql = (
            "SELECT id, name, stripe_customer_id, is_personal "
            "FROM public.organizations "
            "WHERE stripe_customer_id IS NOT NULL "
            "ORDER BY created_at ASC"
        )
        if args.limit:
            sql += f" LIMIT {int(args.limit)}"
        rows = await pool.fetch(sql)
    finally:
        await pool.close()

    logger.info("Processing %d orgs (dry_run=%s, test_mode=%s)",
                len(rows), args.dry_run, args.test_mode)

    counts = {"modified": 0, "skipped-already-tagged": 0, "would-modify": 0, "failed": 0}
    for row in rows:
        outcome = await _process_org(row, args.dry_run)
        print(json.dumps(outcome))
        counts[outcome["outcome"]] = counts.get(outcome["outcome"], 0) + 1

    summary = {"summary": True, "counts": counts, "total": len(rows)}
    print(json.dumps(summary))
    return 1 if counts["failed"] > 0 else 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--dry-run", action="store_true",
                        help="print intended modifications without calling Stripe")
    parser.add_argument("--test-mode", action="store_true",
                        help="use STRIPE_TEST_SECRET_KEY instead of STRIPE_SECRET_KEY")
    parser.add_argument("--limit", type=int, default=0,
                        help="process at most N orgs (0 = unlimited)")
    args = parser.parse_args()

    logging.basicConfig(level=logging.INFO, stream=sys.stderr,
                        format="%(asctime)s %(levelname)s %(message)s")
    return asyncio.run(main_async(args))


if __name__ == "__main__":
    sys.exit(main())
```

import argparse
import asyncio
import json
import logging
import os
import sys

import asyncpg
import stripe

from config import settings


logger = logging.getLogger("verify_stripe_org_metadata")


def _resolve_stripe_key(test_mode: bool) -> str:
    if test_mode:
        key = os.environ.get("STRIPE_TEST_SECRET_KEY", "")
        if not key:
            raise SystemExit("ERROR: --test-mode requires STRIPE_TEST_SECRET_KEY")

async def main_async(args: argparse.Namespace) -> int:

    pool = await asyncpg.create_pool(settings.database_url, min_size=1, max_size=2)
    try:
        rows = await pool.fetch(
            "SELECT id, name, stripe_customer_id FROM public.organizations "
            "WHERE stripe_customer_id IS NOT NULL ORDER BY created_at ASC"

        await pool.close()

    mismatches: list[dict] = []
    for row in rows:
        try:
            cust = stripe.Customer.retrieve(row["stripe_customer_id"])

        "total_checked": len(rows),
        "mismatch_count": len(mismatches),
        "mismatches": mismatches[:25],
    }
    print(json.dumps(summary, indent=2))
    return 1 if mismatches else 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--test-mode", action="store_true")

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| script → Stripe API | Script has STRIPE_SECRET_KEY (live); a bug could modify wrong customers or leak org names to wrong accounts |
| script → public.organizations | Script reads stripe_customer_id values from DB; a corrupted row (wrong cus_id) would tag the wrong customer |
| operator → script | --test-mode flag is the safety guard; default is live mode |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-04-01 | Tampering | Stamp script modifies wrong Stripe customer | mitigate | Script SELECTs stripe_customer_id from DB; DB is source of truth for the mapping; --dry-run lets operator review intended modifications before live run |
| T-12-04-02 | Tampering | Operator runs prod mode by accident | mitigate | --test-mode flag uses separate env var STRIPE_TEST_SECRET_KEY; runbook documents test-mode-first protocol |
| T-12-04-03 | Information Disclosure | Stripe metadata leaks org_name to other dashboards | accept | Stripe Customer metadata is visible only to org's own Stripe account; not surfaced externally |
| T-12-04-04 | Denial of Service | Customer.modify floods customer.updated webhooks | accept | RESEARCH §A2 assumption: Stripe doesn't fire customer.updated for metadata-only changes; if it does, webhook handler is idempotent and rate-limited at FastAPI layer |
| T-12-04-05 | Tampering | Re-running script duplicates work | mitigate | stripe.Customer.retrieve + _is_already_tagged check before modify; "skipped-already-tagged" outcome makes re-runs safe |
| T-12-04-06 | Repudiation | No audit trail of which customers got modified when | mitigate | Per-org JSON line printed to stdout; operator pipes to log file at runbook time |
| T-12-04-07 | Spoofing | Script invoked without correct DATABASE_URL | mitigate | settings.database_url loads from env; asyncpg.create_pool fails fast on bad URL with clear error |
