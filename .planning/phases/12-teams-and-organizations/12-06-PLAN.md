```typescript
/**
 * Phase 12 E2E — full teams-and-orgs happy path.
 *
 * Covers: create org → invite → accept → switch → launch job (mocked) →
 *         org owner views billing → transfer ownership.
 *
 * Requires a running local stack:
 *   - backend on :8000 with settings.organizations_enabled = true
 *   - frontend on :5173 (vite dev)
 *   - Supabase local at port 54321
 *   - Two pre-seeded test users (usera@example.com, userb@example.com)
 *
 * Run: `cd frontend && npx playwright test tests/e2e/organizations.spec.ts`
 */

import { expect, test } from "@playwright/test";

const USER_A_EMAIL = process.env.PHASE12_USER_A_EMAIL ?? "usera-e2e@example.com";
const USER_A_PW = process.env.PHASE12_USER_A_PW ?? "TestPassword123!";
const USER_B_EMAIL = process.env.PHASE12_USER_B_EMAIL ?? "userb-e2e@example.com";
const USER_B_PW = process.env.PHASE12_USER_B_PW ?? "TestPassword123!";
const ORG_NAME = `Acme Bio E2E ${Date.now()}`;

test.describe.serial("Phase 12: full teams flow", () => {
  let inviteToken: string | null = null;

  test.beforeAll(async () => {
    // Test runner pre-seeding (signup or DB seed). Skip if accounts already exist.
  });

  test("1. User A signs in and creates a team org", async ({ page }) => {
    await page.goto("/login");
    await page.fill('input[name="email"]', USER_A_EMAIL);
    await page.fill('input[name="password"]', USER_A_PW);
    await page.click('button[type="submit"]');
    await expect(page).toHaveURL(/\/(chat|jobs|home)/);

    await page.goto("/organizations/new");
    await page.fill("#org-name", ORG_NAME);
    await page.click('button:has-text("Create organization")');

    // After create_org, the context refresh + setActiveOrg triggers a reload.
    // The new org should be active and visible in the header switcher.
    await expect(page.getByText(ORG_NAME)).toBeVisible({ timeout: 10_000 });
  });

  test("2. User A invites User B from the Organization tab", async ({ page, context }) => {
    // re-login (since previous test ended)
    await loginAs(page, USER_A_EMAIL, USER_A_PW);
    await switchToOrg(page, ORG_NAME);

    await page.goto("/settings?tab=organization");
    await page.click('button:has-text("Invitations")');  // or click "Members" then "Invite"
    await page.fill('input[name="email"]', USER_B_EMAIL);
    await page.selectOption('select[name="role"]', "scientist");
    await page.click('button:has-text("Send invitation")');

    // Verify the invitation row appears in pending list
    await expect(page.getByText(USER_B_EMAIL)).toBeVisible({ timeout: 5_000 });

    // Extract the token. Two options:
    //  (a) Query the local DB via a debug endpoint (recommended for E2E)
    //  (b) Read from a Resend-mock outbox if Phase 11 wired one
    // For now, fetch via the backend test endpoint OR direct DB query through fixtures.
    inviteToken = await fetchInviteToken(USER_B_EMAIL, ORG_NAME);
    expect(inviteToken).toBeTruthy();
  });

  test("3. User B accepts the invitation", async ({ page }) => {
    await loginAs(page, USER_B_EMAIL, USER_B_PW);
    await page.goto(`/invitations/accept?token=${inviteToken}`);

    await expect(page.getByText(ORG_NAME)).toBeVisible();
    await expect(page.getByRole("button", { name: /Accept invitation/i })).toBeVisible();
    await page.click('button:has-text("Accept invitation")');

    // After accept, context.setActiveOrg reloads into the new org.
    await expect(page.getByText(ORG_NAME)).toBeVisible({ timeout: 10_000 });
    // User B should now see the Members tab in Settings.
    await page.goto("/settings?tab=organization");
    await expect(page.getByText("Members")).toBeVisible();
  });

  test("4. User A sees User B in the members list", async ({ page }) => {
    await loginAs(page, USER_A_EMAIL, USER_A_PW);
    await switchToOrg(page, ORG_NAME);
    await page.goto("/settings?tab=organization");
    await page.click('button:has-text("Members")');
    await expect(page.getByText(USER_B_EMAIL)).toBeVisible();
  });

  test("5. User B launches a smoke job in the team org", async ({ page }) => {
    await loginAs(page, USER_B_EMAIL, USER_B_PW);
    await switchToOrg(page, ORG_NAME);
    // Walk through the chat-to-launch flow OR call a dev-only /jobs/launch_smoke
    // endpoint that creates a row directly. This test asserts the row appears.
    await page.goto("/jobs");
    // Optional: trigger via API call from within the page context.
    await page.evaluate(async () => {
      await fetch("/api/jobs/launch_smoke", {
        method: "POST",
        credentials: "include",
        headers: { "X-Org-Id": localStorage.getItem("kendrew.activeOrgId") ?? "" },
      });
    });
    await page.reload();
    await expect(page.getByText(/queued|running|complete/i).first()).toBeVisible({ timeout: 10_000 });
  });

  test("6. User A sees User B's job in the org's history", async ({ page }) => {
    await loginAs(page, USER_A_EMAIL, USER_A_PW);
    await switchToOrg(page, ORG_NAME);
    await page.goto("/jobs");
    // Launched-by column should show USER_B_EMAIL.
    await expect(page.getByText(USER_B_EMAIL)).toBeVisible({ timeout: 5_000 });
  });

  test("7. User A views billing as owner", async ({ page }) => {
    await loginAs(page, USER_A_EMAIL, USER_A_PW);
    await switchToOrg(page, ORG_NAME);
    await page.goto("/settings?tab=billing");
    await expect(page.getByRole("button", { name: /Open billing portal/i })).toBeVisible({ timeout: 5_000 });
  });

  test("8. User B sees the non-owner billing gate", async ({ page }) => {
    await loginAs(page, USER_B_EMAIL, USER_B_PW);
    await switchToOrg(page, ORG_NAME);
    await page.goto("/settings?tab=billing");
    await expect(page.getByText(/Billing is managed by your organization owner/i)).toBeVisible();
    await expect(page.getByText(USER_A_EMAIL)).toBeVisible();
  });

  test("9. User A transfers ownership to User B", async ({ page }) => {
    await loginAs(page, USER_A_EMAIL, USER_A_PW);
    await switchToOrg(page, ORG_NAME);
    await page.goto("/settings?tab=organization");
    await page.click('button:has-text("Members")');
    await page.click('button:has-text("Transfer ownership")');
    await page.selectOption('select[name="target_user"]', { label: USER_B_EMAIL });
    await page.selectOption('select[name="new_self_role"]', "scientist");
    await page.click('button:has-text("Confirm transfer")');
    await expect(page.getByText(/Transferred|now an owner/i)).toBeVisible({ timeout: 5_000 });
  });

  test("10. User B is now the owner", async ({ page }) => {
    await loginAs(page, USER_B_EMAIL, USER_B_PW);
    await switchToOrg(page, ORG_NAME);
    await page.goto("/settings?tab=billing");
    await expect(page.getByRole("button", { name: /Open billing portal/i })).toBeVisible({ timeout: 5_000 });
  });
});

async function loginAs(page: any, email: string, password: string) {
  await page.goto("/login");
  await page.fill('input[name="email"]', email);
  await page.fill('input[name="password"]', password);
  await page.click('button[type="submit"]');
  await expect(page).toHaveURL(/\/(chat|jobs|home)/);
}

async function switchToOrg(page: any, orgName: string) {
  // Click the header switcher and select the named org. Falls back to localStorage write + reload
  // if the dropdown is hidden (e.g., user has only the personal org pre-switch).
  await page.goto("/");
  const switcher = page.locator('button:has-text("' + orgName + '")');
  if (!(await switcher.isVisible({ timeout: 1000 }).catch(() => false))) {
    await page.evaluate((name) => {
      // localStorage.setItem("kendrew.activeOrgId", lookup-from-orgs-mine);
      // Simpler: navigate to switcher path
    }, orgName);
  }
}

async function fetchInviteToken(email: string, orgName: string): Promise<string | null> {

  //  (b) Direct DB query via Playwright's request fixture
  // Use (a) if present; otherwise (b). For Phase 12 E2E, (a) is the simpler hook.
  return null;  // implementor wires this up against the actual debug endpoint
}
```
