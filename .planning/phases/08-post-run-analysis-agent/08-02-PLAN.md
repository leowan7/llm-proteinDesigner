    ```python
    """BioPython-based PDB structural feature extraction for post-run analysis.

    Computes:
    - Buried Surface Area (BSA): SASA(chain_A) + SASA(chain_B) - SASA(complex)
    - Inter-chain clash count: atom pairs closer than sum of VdW radii
    - Interface contact count: residues within distance cutoff

    All functions accept a PDB file path and chain IDs. Returns None for
    individual metrics on failure (malformed PDB, missing chain) rather than
    raising — the agent handles None gracefully.
    """

    import logging
    from pathlib import Path

    import numpy as np
    from Bio.PDB import PDBParser, NeighborSearch
    from Bio.PDB.Polypeptide import is_aa
    from Bio.PDB.SASA import ShrakeRupley

    logger = logging.getLogger(__name__)

    # Van der Waals radii (Angstroms) — simplified set covering standard amino acid atoms.
    # Source: CHARMM36 force field, rounded to 2 decimal places.
    VDW_RADII: dict[str, float] = {
        "C": 1.70, "N": 1.55, "O": 1.52, "S": 1.80, "H": 1.20,
        "P": 1.80, "SE": 1.90, "FE": 1.50, "ZN": 1.39, "MG": 1.73,
    }
    _DEFAULT_VDW = 1.50
    _CLASH_TOLERANCE = 0.4  # Angstroms of overlap tolerance


    def compute_bsa(pdb_path: str, chain_a: str, chain_b: str) -> float | None:
        """Compute buried surface area between two chains.

        BSA = SASA(chain_A alone) + SASA(chain_B alone) - SASA(complex)

        Args:
            pdb_path: Path to PDB file.
            chain_a: First chain ID.
            chain_b: Second chain ID.

        Returns:
            BSA in Angstrom^2, or None if computation fails.
        """
        try:
            parser = PDBParser(QUIET=True)
            structure = parser.get_structure("complex", pdb_path)
            model = structure[0]

            if chain_a not in [c.id for c in model] or chain_b not in [c.id for c in model]:
                return None

            sr = ShrakeRupley(probe_radius=1.4, n_points=100)

            # SASA of entire complex
            sr.compute(model, level="A")
            complex_sasa = sum(atom.sasa for chain in model for atom in chain.get_atoms())

            # SASA of chain A alone — detach chain B temporarily
            # Simpler approach: compute per-chain SASA by isolating atoms
            chain_a_atoms = list(model[chain_a].get_atoms())
            chain_b_atoms = list(model[chain_b].get_atoms())

            # Re-parse for isolated chains (cleanest approach)
            sr_a = ShrakeRupley(probe_radius=1.4, n_points=100)
            sr_b = ShrakeRupley(probe_radius=1.4, n_points=100)

            # Create temporary structures for each chain
            from Bio.PDB import StructureBuilder
            import copy

            # Deep copy to avoid modifying original
            struct_a = parser.get_structure("chain_a", pdb_path)
            model_a = struct_a[0]
            chains_to_remove = [c.id for c in model_a if c.id != chain_a]
            for cid in chains_to_remove:
                model_a.detach_child(cid)
            sr_a.compute(model_a, level="A")
            sasa_a = sum(atom.sasa for atom in model_a.get_atoms())

            struct_b = parser.get_structure("chain_b", pdb_path)
            model_b = struct_b[0]
            chains_to_remove = [c.id for c in model_b if c.id != chain_b]
            for cid in chains_to_remove:
                model_b.detach_child(cid)
            sr_b.compute(model_b, level="A")
            sasa_b = sum(atom.sasa for atom in model_b.get_atoms())

            bsa = sasa_a + sasa_b - complex_sasa
            return round(max(0, bsa), 1)
        except Exception:
            logger.exception("BSA computation failed for %s", pdb_path)
            return None


    def count_clashes(pdb_path: str, chain_a: str, chain_b: str) -> int | None:
        """Count inter-chain steric clashes (atom pairs closer than sum of VdW radii).

        Args:
            pdb_path: Path to PDB file.
            chain_a: First chain ID.
            chain_b: Second chain ID.

        Returns:
            Number of clashing atom pairs, or None if computation fails.
        """
        try:
            parser = PDBParser(QUIET=True)
            structure = parser.get_structure("complex", pdb_path)
            model = structure[0]

            if chain_a not in [c.id for c in model] or chain_b not in [c.id for c in model]:
                return None

            atoms_a = [a for a in model[chain_a].get_atoms() if a.element != "H"]
            atoms_b = [a for a in model[chain_b].get_atoms() if a.element != "H"]

            if not atoms_a or not atoms_b:
                return 0

            ns = NeighborSearch(atoms_b)
            clashes = 0

            for atom_a in atoms_a:
                radius_a = VDW_RADII.get(atom_a.element.upper(), _DEFAULT_VDW)
                search_radius = radius_a + max(VDW_RADII.values())
                nearby = ns.search(atom_a.coord, search_radius)
                for atom_b in nearby:
                    radius_b = VDW_RADII.get(atom_b.element.upper(), _DEFAULT_VDW)
                    dist = atom_a - atom_b
                    if dist < (radius_a + radius_b - _CLASH_TOLERANCE):
                        clashes += 1

            return clashes // 2  # Each pair counted from both directions

        except Exception:
            logger.exception("Clash count failed for %s", pdb_path)
            return None


    def count_interface_contacts(
        pdb_path: str, chain_a: str, chain_b: str, cutoff: float = 5.0
    ) -> int | None:
        """Count residues on chain_a within cutoff distance of chain_b.

        Reuses the same spatial query approach as pdb_utils/interface.py.

        Args:
            pdb_path: Path to PDB file.
            chain_a: Chain to count contacts on.
            chain_b: Partner chain.
            cutoff: Distance cutoff in Angstroms.

        Returns:
            Number of interface residues on chain_a, or None on failure.
        """
        try:
            parser = PDBParser(QUIET=True)
            structure = parser.get_structure("complex", pdb_path)
            model = structure[0]

            if chain_a not in [c.id for c in model] or chain_b not in [c.id for c in model]:
                return None

            atoms_b = [
                atom for residue in model[chain_b]
                if is_aa(residue, standard=True)
                for atom in residue if atom.element != "H"
            ]

            if not atoms_b:
                return 0

            ns = NeighborSearch(atoms_b)
            contact_residues = set()

            for residue in model[chain_a]:
                if not is_aa(residue, standard=True):
                    continue
                for atom in residue:
                    if atom.element == "H":
                        continue
                    if ns.search(atom.coord, cutoff):
                        contact_residues.add(residue.get_id()[1])
                        break  # one contact per residue is enough

            return len(contact_residues)

        except Exception:
            logger.exception("Contact count failed for %s", pdb_path)
            return None


    def extract_structural_features(
        pdb_path: str, chain_a: str, chain_b: str
    ) -> dict:
        """Extract all structural features from a two-chain PDB.

        Convenience function that runs all three analyses. Individual failures
        return None for that metric without blocking others.

        Args:
            pdb_path: Path to PDB file.
            chain_a: Target chain ID.
            chain_b: Binder chain ID.

        Returns:
            Dict with keys: bsa (float|None), clash_count (int|None),
            interface_contacts (int|None).
        """
        return {
            "bsa": compute_bsa(pdb_path, chain_a, chain_b),
            "clash_count": count_clashes(pdb_path, chain_a, chain_b),
            "interface_contacts": count_interface_contacts(pdb_path, chain_a, chain_b),
        }
    ```
