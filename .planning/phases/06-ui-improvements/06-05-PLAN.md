    **User API client** `frontend/src/lib/user.ts`:

    ```typescript
    import { api } from "./api";

    const API_BASE = "http://localhost:8000";

    export interface UserSettings {
      email: string;
      display_name: string;
      notification_preferences: {
        job_complete: boolean;
        job_failure: boolean;
      };
    }

    export interface UsageData {
      period_start: string;
      job_count: number;
      total_spend_usd: number;
      recent_charges: Array<{
        id: string;
        name: string | null;
        tool: string | null;
        completed_at: string;
        gpu_cost_usd: number;
      }>;
    }

    export interface PaymentMethod {
      has_payment_method: boolean;
      brand?: string;
      last4?: string;
      exp_month?: number;
      exp_year?: number;
    }

    export async function getSettings(): Promise<UserSettings> {
      return api<UserSettings>("/user/settings", { method: "GET" });
    }

    export async function updateSettings(data: {
      display_name?: string;
      notification_preferences?: { job_complete: boolean; job_failure: boolean };
    }): Promise<void> {
      await api("/user/settings", {
        method: "PUT",
        body: JSON.stringify(data),
      });
    }

    export async function getUsage(): Promise<UsageData> {
      return api<UsageData>("/user/usage", { method: "GET" });
    }

    export async function getPaymentMethod(): Promise<PaymentMethod> {
      return api<PaymentMethod>("/billing/payment-method", { method: "GET" });
    }

    export async function createPortalSession(returnUrl: string): Promise<string> {
      const data = await api<{ url: string }>("/billing/portal", {
        method: "POST",
        body: JSON.stringify({ return_url: returnUrl }),
      });
      return data.url;
    }
    ```

    **aria-live for chat messages** per D-24 and research pitfall 7:

    In `MessageList.tsx`, do NOT place `aria-live` on the full message container (this would announce all messages on mount). Instead, add a visually hidden announcement div:
    ```tsx
    {/* Screen reader announcement for new messages only */}
    <div
      role="status"
      aria-live="polite"
      aria-atomic="true"
      className="sr-only"
    >
      {lastAnnouncedMessage}
    </div>
    ```
    Where `lastAnnouncedMessage` is a state/prop that updates only when a NEW message arrives (not on initial load). Pass it as a prop from ChatPage: when a new assistant message completes (after `done` SSE event), set `lastAnnouncedMessage` to a truncated version of the message text (first 100 chars).

    **aria-live for job status SSE updates** per D-24:

    In `JobStatusCard.tsx` or `JobPage.tsx`, add an assertive live region for status changes:
    ```tsx
    <div aria-live="assertive" aria-atomic="true" className="sr-only">
      {statusAnnouncementText}
    </div>
    ```
    Where `statusAnnouncementText` updates when the job status changes (e.g., "Job status changed to running", "Job status changed to complete").
