    **Session API client** `frontend/src/lib/sessions.ts`:

    ```typescript
    import { api } from "./api";

    export interface SessionSummary {
      id: string;
      title: string | null;
      created_at: string;
      updated_at: string;
    }

    export interface SessionDetail {
      id: string;
      title: string | null;
      agent_history: unknown[];
      messages: Array<{
        id: string;
        role: "user" | "assistant";
        content: string;
        cards: unknown[] | null;
        sort_order: number;
      }>;
    }

    export async function listSessions(limit = 50, before?: string): Promise<{sessions: SessionSummary[]}> {
      const params = new URLSearchParams({ limit: String(limit) });
      if (before) params.set("before", before);
      return api<{sessions: SessionSummary[]}>(`/sessions?${params}`, { method: "GET" });
    }

    export async function loadSession(sessionId: string): Promise<SessionDetail> {
      return api<SessionDetail>(`/sessions/${sessionId}`, { method: "GET" });
    }

    export async function createPersistentSession(): Promise<SessionSummary> {
      return api<SessionSummary>("/sessions", { method: "POST" });
    }

    export async function deleteSessionApi(sessionId: string): Promise<void> {
      await api(`/sessions/${sessionId}`, { method: "DELETE" });
    }

    export async function updateSessionTitle(sessionId: string, title: string): Promise<void> {
      await api(`/sessions/${sessionId}`, { method: "PUT", body: JSON.stringify({ title }) });
    }
    ```
