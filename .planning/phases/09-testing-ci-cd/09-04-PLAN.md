Create `.github/workflows/test.yml` with 4 jobs per D-08. Use `supabase/setup-cli@v1` for the real DB per D-02/D-10.

```yaml
name: Test Suite

on:
  pull_request:
    branches: [main, master]

# Cancel previous runs on same PR
concurrency:
  group: test-${{ github.head_ref }}
  cancel-in-progress: true

jobs:
  backend:
    name: Backend Tests
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: backend
    steps:
      - uses: actions/checkout@v4

      - uses: supabase/setup-cli@v1
        with:
          version: latest

      - name: Start Supabase
        run: supabase start
        working-directory: .

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - uses: actions/cache@v4
        with:
          path: ~/.cache/pip
          key: pip-${{ hashFiles('backend/requirements.txt') }}
          restore-keys: pip-

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Run tests with coverage
        run: python -m pytest tests/ --cov=. --cov-fail-under=80 --cov-report=xml --cov-report=term-missing
        env:
          TESTING: "true"
          SUPABASE_URL: http://127.0.0.1:54321
          SUPABASE_SERVICE_KEY: ${{ secrets.SUPABASE_SERVICE_KEY }}
          COOKIE_SECURE: "false"
          CSRF_SECRET: "ci-csrf-secret"

      - name: Upload coverage artifact
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: backend-coverage
          path: backend/coverage.xml

  frontend-unit:
    name: Frontend Unit Tests
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: frontend
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      - uses: actions/cache@v4
        with:
          path: frontend/node_modules
          key: node-${{ hashFiles('frontend/package-lock.json') }}
          restore-keys: node-

      - name: Install dependencies
        run: npm ci

      - name: Run Vitest
        run: npm run test

  e2e:
    name: E2E Tests
    runs-on: ubuntu-latest
    needs: [backend, frontend-unit]
    steps:
      - uses: actions/checkout@v4

      - uses: supabase/setup-cli@v1
        with:
          version: latest

      - name: Start Supabase
        run: supabase start

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - uses: actions/cache@v4
        with:
          path: ~/.cache/pip
          key: pip-${{ hashFiles('backend/requirements.txt') }}
          restore-keys: pip-

      - name: Install backend deps
        run: pip install -r backend/requirements.txt

      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      - uses: actions/cache@v4
        with:
          path: frontend/node_modules
          key: node-${{ hashFiles('frontend/package-lock.json') }}
          restore-keys: node-

      - name: Install frontend deps
        run: cd frontend && npm ci

      - name: Install Playwright browsers
        run: cd frontend && npx playwright install --with-deps chromium

      - name: Start backend
        run: |
          cd backend && uvicorn main:app --host 0.0.0.0 --port 8000 &
          sleep 3
        env:
          TESTING: "true"
          SUPABASE_URL: http://127.0.0.1:54321
          SUPABASE_SERVICE_KEY: ${{ secrets.SUPABASE_SERVICE_KEY }}
          COOKIE_SECURE: "false"
          CSRF_SECRET: "ci-csrf-secret"

      - name: Run Playwright tests
        run: cd frontend && npx playwright test
        env:
          CI: "true"

      - name: Upload Playwright report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          path: frontend/playwright-report/

  lint:
    name: Lint & Type Check
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install ruff
        run: pip install ruff

      - name: Lint Python
        run: ruff check backend/

      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      - uses: actions/cache@v4
        with:
          path: frontend/node_modules
          key: node-${{ hashFiles('frontend/package-lock.json') }}
          restore-keys: node-

      - name: Install frontend deps
        run: cd frontend && npm ci

      - name: TypeScript type check
        run: cd frontend && npx tsc --noEmit

      - name: ESLint (with jsx-a11y)
        run: cd frontend && npx eslint .
```

Key design choices per D-08:
- All 4 jobs must pass for PR to merge (GitHub branch protection enforces this — configured separately by user)
- `concurrency` group cancels stale runs on same PR to save CI minutes
- `e2e` job depends on backend + frontend-unit (runs only after unit tests pass — saves compute on obvious failures)
- `lint` job runs independently in parallel with backend and frontend-unit
- Coverage XML uploaded as artifact for optional Codecov integration later
- Supabase CLI started in both backend and e2e jobs (real DB per D-02)
- SUPABASE_SERVICE_KEY from GitHub Secrets per D-10

Create `.github/workflows/smoke.yml` per D-11, D-12, D-13. This is a separate workflow triggered manually or after deploy completes.

```yaml
name: Post-Deploy Smoke Test

on:
  workflow_dispatch:
    inputs:
      environment:
        description: "Target environment URL (e.g., https://kendrew.app)"
        required: true
        type: string

jobs:
  smoke:
    name: Smoke Test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: "Check 1: Health endpoint returns 200"
        run: |
          STATUS=$(curl -s -o /dev/null -w "%{http_code}" "${{ inputs.environment }}/health")
          if [ "$STATUS" != "200" ]; then
            echo "FAIL: Health endpoint returned $STATUS"
            exit 1
          fi
          echo "PASS: Health endpoint returned 200"

      - name: "Check 2: Health endpoint response time < 2s"
        run: |
          TIME=$(curl -s -o /dev/null -w "%{time_total}" "${{ inputs.environment }}/health")
          echo "Response time: ${TIME}s"
          if (( $(echo "$TIME > 2.0" | bc -l) )); then
            echo "FAIL: Response time ${TIME}s exceeds 2s threshold"
            exit 1
          fi
          echo "PASS: Response time within threshold"

      - name: "Check 3: Auth flow works (login with test account)"
        run: |
          RESP=$(curl -s -w "\n%{http_code}" -X POST \
            "${{ inputs.environment }}/auth/login" \
            -H "Content-Type: application/json" \
            -d '{"email":"test@example.com","password":"Password123!"}')
          STATUS=$(echo "$RESP" | tail -1)
          if [ "$STATUS" != "200" ]; then
            echo "FAIL: Login returned $STATUS"
            exit 1
          fi
          echo "PASS: Auth login returned 200"

      - name: "Check 4: Frontend loads"
        uses: actions/setup-node@v4
        with:
          node-version: "20"

      - run: npm install -g playwright && npx playwright install chromium
      
      - name: Verify frontend renders
        run: |
          npx playwright test --config=/dev/null -e "
          const { chromium } = require('playwright');
          (async () => {
            const browser = await chromium.launch();
            const page = await browser.newPage();
            await page.goto('${{ inputs.environment }}/login', { timeout: 15000 });
            const title = await page.title();
            console.log('Page title:', title);
            if (!title) {
              console.error('FAIL: Frontend did not render (empty title)');
              process.exit(1);
            }
            console.log('PASS: Frontend loaded successfully');
            await browser.close();
          })();
          " 2>/dev/null || node -e "
          const { chromium } = require('playwright');
          (async () => {
            const browser = await chromium.launch();
            const page = await browser.newPage();
            await page.goto('${{ inputs.environment }}/login', { timeout: 15000 });
            const title = await page.title();
            if (!title) { process.exit(1); }
            console.log('PASS: Frontend loaded');
            await browser.close();
          })();"

      - name: Alert on failure
        if: failure()
        run: |
          echo "::error::Smoke test failed for ${{ inputs.environment }}. Manual rollback may be needed via Railway/Vercel deploy history."
          echo "Check Sentry for error details."
```

Key design choices per D-11/D-12/D-13:
- `workflow_dispatch` trigger — user manually runs after deploy (per D-13). Can also be triggered via GitHub API from a deploy workflow later.
- 4 checks match D-11: health 200, response time <2s, auth works, frontend loads
- No automated rollback per D-12 — the failure step alerts and recommends manual rollback via Railway/Vercel deploy history
- Uses Sentry as the alerting channel per D-12 (the `::error::` annotation creates a GitHub notification)
- Frontend check uses Playwright to actually render the page (not just curl for HTML) — validates that JS loads and the React app initializes

NOTE: The frontend check step is intentionally simple. A more robust version would use a standalone Playwright script file, but for a smoke test, an inline check is sufficient. If it becomes flaky, refactor to a dedicated `e2e/smoke.spec.ts` file.

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-06 | Information Disclosure | CI secrets | mitigate | Use ${{ secrets.* }} syntax which GitHub masks in logs; never echo secrets directly |
| T-09-07 | Information Disclosure | smoke test credentials | accept | test@example.com is a seed account with no real data; acceptable risk for smoke testing |
| T-09-08 | Elevation of Privilege | CI workflow permissions | mitigate | Workflows use default read-only permissions; no write access to repo from PR workflows |
