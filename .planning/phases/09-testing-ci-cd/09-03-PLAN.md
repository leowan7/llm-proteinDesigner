3. Create page object files in `frontend/e2e/pages/`:

   **LoginPage.ts:**
   ```typescript
   import { Page } from "@playwright/test";
   
   export class LoginPage {
     constructor(private page: Page) {}
   
     async goto() {
       await this.page.goto("/login");
     }
   
     async login(email: string, password: string) {
       await this.page.goto("/login");
       await this.page.fill('[name="email"]', email);
       await this.page.fill('[name="password"]', password);
       await this.page.click('button[type="submit"]');
       await this.page.waitForURL("/", { timeout: 10000 });
     }
   
     async getErrorMessage() {
       return this.page.textContent('[role="alert"]');
     }
   }
   ```
   Read Login.tsx to verify the exact input names and button selectors. Adjust selectors to match the actual DOM structure.

   **ChatPage.ts:**
   - `async goto()` — navigate to /
   - `async sendMessage(text: string)` — fill chat input (read ChatPage to find the textarea selector), click send button
   - `async waitForAgentResponse()` — wait for a message bubble from the agent to appear (use `page.waitForSelector` with a selector for agent messages)
   - `async getLastMessage()` — return text content of last message element

   **JobPage.ts:**
   - `async goto(jobId: string)` — navigate to /jobs/{jobId}
   - `async getStatus()` — return text content of the status badge element
   - `async waitForStatus(status: string)` — poll until status text matches

   **SettingsPage.ts:**
   - `async goto()` — navigate to /settings
   - `async getActiveTab()` — return the text of the currently active tab
   - `async clickTab(name: string)` — click a tab by name

1. Create `frontend/e2e/auth.spec.ts` (D-06 flow 1: Login + auth):
   ```typescript
   import { test, expect } from "@playwright/test";
   import { LoginPage } from "./pages/LoginPage";
   
   test.describe("Authentication", () => {
     test("login with valid credentials redirects to home", async ({ page }) => {
       const loginPage = new LoginPage(page);
       await loginPage.login("test@example.com", "Password123!");
       await expect(page).toHaveURL("/");
     });
   
     test("login with invalid credentials shows error", async ({ page }) => {
       const loginPage = new LoginPage(page);
       await page.goto("/login");
       await page.fill('[name="email"]', "wrong@example.com");
       await page.fill('[name="password"]', "wrongpassword");
       await page.click('button[type="submit"]');
       // Should stay on login page and show error
       await expect(page).toHaveURL(/login/);
       await expect(page.locator('[role="alert"], .text-red, .text-destructive')).toBeVisible({ timeout: 5000 });
     });
   
     test("unauthenticated user is redirected to login", async ({ page }) => {
       await page.goto("/");
       await expect(page).toHaveURL(/login/);
     });
   
     test("session persists across page refresh", async ({ page }) => {
       const loginPage = new LoginPage(page);
       await loginPage.login("test@example.com", "Password123!");
       await expect(page).toHaveURL("/");
       await page.reload();
       // Should still be on home, not redirected to login
       await expect(page).toHaveURL("/");
     });
   });
   ```
   Adjust selectors after reading the page objects created in Task 1.

2. Create `frontend/e2e/chat.spec.ts` (D-06 flow 2: Chat + job launch):

     - Wait for agent response with `page.waitForSelector` targeting the agent message element
     - Assert at least one agent message is visible

   NOTE: Mark chat tests with `test.slow()` — agent responses via Anthropic API take 5-15s. In CI, these may need mocking or may be skipped with `test.skip(process.env.CI)` if no API key is available.
